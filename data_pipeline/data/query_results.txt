QUERY 5 - Top-rated books with categories
============================================================
SELECT
    b.title,
    b.price_inr,
    b.rating,
    c.category_name
FROM books b
LEFT JOIN categories c
    ON b.category_id = c.category_id
ORDER BY b.rating DESC, b.book_id ASC
LIMIT 10
